
set(CUACC_CORE core CACHE PATH "cuacc main dir")
set(CUACC_CORE_DIR ${CMAKE_CURRENT_LIST_DIR} CACHE PATH "cuacc main dir")
set(CUACC_CORE_SRC_DIR ${CUACC_CORE_DIR}/src CACHE PATH "cuacc src dir")
set(CUACC_CORE_INCLUDE_DIR ${CUACC_CORE_DIR}/include CACHE PATH "cuacc include dir")
set(CUACC_CORE_BUILD_DIR ${CMAKE_CURRENT_BINARY_DIR} CACHE PATH "cuacc build directory")

if(UNIX)
    if("${CMAKE_BUILD_TYPE}" STREQUAL "")
        set(CMAKE_BUILD_TYPE RELEASE)
    else()
        string(TOUPPER ${CMAKE_BUILD_TYPE} CMAKE_BUILD_TYPE)
    endif()
endif()

file(GLOB_RECURSE CUACC_CORE_CXX_SRC
    ${CUACC_CORE_SRC_DIR}/*.cpp
)
file(GLOB_RECURSE CUACC_CORE_CXX_HEADERS
    ${CUACC_CORE_SRC_DIR}/*.hpp
    ${CUACC_CORE_INCLUDE_DIR}/*.hpp
)

add_library(${CUACC_CORE} STATIC ${CUACC_CORE_CXX_SRC} ${CUACC_CORE_CU_SRC})
add_library(${LIB_CUACC}::${CUACC_CORE} ALIAS ${CUACC_CORE})

target_link_libraries(${CUACC_CORE} PUBLIC
)

target_include_directories(${CUACC_CORE} PUBLIC
    $<BUILD_INTERFACE:${CUACC_CORE_INCLUDE_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/>
)

target_compile_options(${CUACC_CORE} PRIVATE
    $<$<CONFIG:RELEASE>: ${CUACC_REL_BUILD_FLAGS} ${CUACC_STD_LIB_CXX_FLAGS} ${CUACC_REL_WARN_FLAGS} $<$<BOOL:${CUACC_TCMALLOC}>:${CUACC_TCMALLOC_FLAGS}>>
    $<$<CONFIG:DEBUG>: ${CUACC_DBG_BUILD_FLAGS} ${CUACC_STD_LIB_CXX_FLAGS} ${CUACC_DBG_WARN_FLAGS} $<$<BOOL:${CUACC_TCMALLOC}>:${CUACC_TCMALLOC_FLAGS}>>
)

target_link_options(${CUACC_CORE} PUBLIC
    $<$<CONFIG:RELEASE>: -s ${CUACC_STD_LIB_CXX_LINKER_FLAGS} $<$<BOOL:${CUACC_TCMALLOC}>:${CUACC_TCMALLOC_LINKER_FLAGS}>>
    $<$<CONFIG:DEBUG>: ${CUACC_STD_LIB_CXX_LINKER_FLAGS} $<$<BOOL:${CUACC_TCMALLOC}>:${CUACC_TCMALLOC_LINKER_FLAGS}>>
)

set_target_properties(${CUACC_CORE} PROPERTIES
    INTERPROCEDURAL_OPTIMIZATION ON
    CXX_STANDARD ${CUACC_CXX_STANDARD}
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF
    OUTPUT_NAME ${LIB_CUACC}-${CUACC_CORE}
    # VERSION ${PROJECT_VERSION}
    SOVERSION ${PROJECT_VERSION}
    # SOVERSION ${PROJECT_VERSION_MAJOR}
)

if(CUACC_INSTALL)
    install(
        TARGETS ${CUACC_CORE}
        EXPORT ${LIB_CUACC}_Targets
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        NAMELINK_SKIP
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} # for windows
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
        INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
    )

    install(TARGETS ${CUACC_CORE}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        NAMELINK_ONLY
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} # for windows
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    )

    install(
        DIRECTORY ${CUACC_CORE_INCLUDE_DIR}/
        DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/${CUACC_CORE}
        FILES_MATCHING
        PATTERN "*.hpp"
        REGEX "/[^.]+$"
    )
endif(CUACC_INSTALL)
