
set(CUACC_MEMORY memory CACHE PATH "cuacc main dir")
set(CUACC_MEMORY_DIR ${CMAKE_CURRENT_LIST_DIR} CACHE PATH "cuacc main dir")
set(CUACC_MEMORY_SRC_DIR ${CUACC_MEMORY_DIR}/src CACHE PATH "cuacc src dir")
set(CUACC_MEMORY_INCLUDE_DIR ${CUACC_MEMORY_DIR}/include CACHE PATH "cuacc include dir")
set(CUACC_MEMORY_BUILD_DIR ${CMAKE_CURRENT_BINARY_DIR} CACHE PATH "cuacc build directory")

if(UNIX)
    if("${CMAKE_BUILD_TYPE}" STREQUAL "")
        set(CMAKE_BUILD_TYPE RELEASE)
    else()
        string(TOUPPER ${CMAKE_BUILD_TYPE} CMAKE_BUILD_TYPE)
    endif()
endif()

file(GLOB_RECURSE CUACC_MEMORY_CXX_SRC
    ${CUACC_MEMORY_SRC_DIR}/*.cpp
)
file(GLOB_RECURSE CUACC_MEMORY_CXX_HEADERS
    ${CUACC_MEMORY_SRC_DIR}/*.hpp
    ${CUACC_MEMORY_INCLUDE_DIR}/*.hpp
)

add_library(${CUACC_MEMORY} INTERFACE ${CUACC_MEMORY_CXX_SRC} ${CUACC_MEMORY_CXX_HEADERS})
add_library(${LIB_CUACC}::${CUACC_MEMORY} ALIAS ${CUACC_MEMORY})

target_link_libraries(${CUACC_MEMORY} INTERFACE
    CUDA::cudart_static
    CUDA::cuda_driver
)

target_include_directories(${CUACC_MEMORY} INTERFACE
    $<BUILD_INTERFACE:${CUACC_MEMORY_INCLUDE_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/>
)

set_target_properties(${CUACC_MEMORY} PROPERTIES
    INTERPROCEDURAL_OPTIMIZATION ON
    CXX_STANDARD ${CUACC_CXX_STANDARD}
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF
    OUTPUT_NAME ${LIB_CUACC}-${CUACC_MEMORY}
    # VERSION ${PROJECT_VERSION}
    SOVERSION ${PROJECT_VERSION}
    # SOVERSION ${PROJECT_VERSION_MAJOR}
)

if(CUACC_INSTALL)
    install(
        TARGETS ${CUACC_MEMORY}
        EXPORT ${LIB_CUACC}_Targets
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        NAMELINK_SKIP
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} # for windows
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
        INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
    )

    install(TARGETS ${CUACC_MEMORY}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        NAMELINK_ONLY
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} # for windows
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    )

    install(
        DIRECTORY ${CUACC_MEMORY_INCLUDE_DIR}/
        DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/${CUACC_MEMORY}

        FILES_MATCHING
        PATTERN "*.hpp"
        REGEX "/[^.]+$"
    )
endif(CUACC_INSTALL)
