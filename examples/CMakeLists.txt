set(CUACC_VEC_ADD ${LIB_CUACC}-vecadd CACHE INTERNAL "")
set(CUACC_CUDA_FP16 ${LIB_CUACC}-cuda-fp16 CACHE INTERNAL "")
set(CUACC_CUDA_DEVICE ${LIB_CUACC}-cuda-device CACHE INTERNAL "")

project(
    "${LIB_CUACC}-examples"
    VERSION 0.1.0
    DESCRIPTION "acceleration base library on top of cuda in C++ examples and usage."
    LANGUAGES CXX
)

if(UNIX)
    list(APPEND CMAKE_INSTALL_RPATH $<$<BOOL:${CUACC_SHARED_LIB}>:${CMAKE_INSTALL_FULL_LIBDIR}>)

    if(APPLE)
        set(CMAKE_MACOSX_RPATH ON)
    endif()
endif()

find_package(CUDAToolkit REQUIRED)

file(GLOB_RECURSE CUACC_VEC_ADD_SOURCES
    ${CMAKE_CURRENT_LIST_DIR}/vecadd.cpp
)
file(GLOB_RECURSE CUACC_CUDA_FP16_SOURCES
    ${CMAKE_CURRENT_LIST_DIR}/cuda_fp16.cpp
)
file(GLOB_RECURSE CUACC_CUDA_DEVICE_SOURCES
    ${CMAKE_CURRENT_LIST_DIR}/cudaDevice.cpp
)

add_executable(${CUACC_VEC_ADD}
    ${CUACC_VEC_ADD_SOURCES}
)
target_link_libraries(${CUACC_VEC_ADD} PRIVATE
    cuacc::kernels
    cuacc::core
    cuacc::memory
)
target_compile_options(${CUACC_VEC_ADD} PRIVATE
    $<$<CONFIG:RELEASE>: ${CUACC_REL_BUILD_FLAGS} ${CUACC_STD_LIB_CXX_FLAGS} ${CUACC_REL_WARN_FLAGS} $<$<BOOL:${CUACC_TCMALLOC}>:${CUACC_TCMALLOC_FLAGS}>>
    $<$<CONFIG:DEBUG>: ${CUACC_DBG_BUILD_FLAGS} ${CUACC_STD_LIB_CXX_FLAGS} ${CUACC_DBG_WARN_FLAGS} $<$<BOOL:${CUACC_TCMALLOC}>:${CUACC_TCMALLOC_FLAGS}>>
)
target_link_options(${CUACC_VEC_ADD} PRIVATE
    $<$<CONFIG:RELEASE>: -s ${CUACC_STD_LIB_CXX_LINKER_FLAGS} $<$<BOOL:${CUACC_TCMALLOC}>:${CUACC_TCMALLOC_LINKER_FLAGS}>>
    $<$<CONFIG:DEBUG>: ${CUACC_STD_LIB_CXX_LINKER_FLAGS} $<$<BOOL:${CUACC_TCMALLOC}>:${CUACC_TCMALLOC_LINKER_FLAGS}>>
)
set_target_properties(${CUACC_VEC_ADD} PROPERTIES
    INTERPROCEDURAL_OPTIMIZATION ON
    CXX_STANDARD ${CUACC_CXX_STANDARD}
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF
)
